# 1. 不重复随机整数 — 控制抽题编号
def Random_num(num_max, num_qty):
    num_list = []
    while len(num_list) < num_qty:
        num = random.randint(2, num_max)
        if num not in num_list:
            num_list.append(num)
    return num_list
